# EC engine routines (sub-CPU addresses), used by ecdis.py to label calls.
# Task control block (a4): +0 next task, +4 resume PC, +8 sleep counter, +A return PC.
# Verified by reading SUB.BIN; see docs/re_notes.md.
0x68D6  sleep            # d0 = frames; resume after the call
0x68D0  yield            # resume after the call next frame
0x68CA  end_frame        # jmp only: resume at last mark next frame
0x6960  mark             # set resume point here, keep running
0x6A1A  halt             # sleep forever
0x6A30  get_halt_addr    # d0 = address of halt (for set_task_code)
0x68B8  set_task1_code   # next task in chain: resume at d0
0x68BE  set_task2_code   # 2nd next task
0x68C4  set_task3_code   # 3rd next task
0x68B2  task_tail        # walk to end of task chain
0x68E8  gosub            # run code at a0; it returns via TCB+A
0x6BB4  spawn_object     # d0 graphic id, d1 x, d2 y, a1 code
0x6966  load_resources   # inline ids, $FFFF-terminated
0x696C  load_resources2  # inline ids, $FFFF-terminated (keeps a4)
0x69FC  set_palette      # a0 colors, d0..d1 palette index range
0x6D58  update_if_active # if flag $1A3D(a6) bit1: call $47F7E
